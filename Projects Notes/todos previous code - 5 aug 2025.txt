  <script>

    let todos = [];
    let userId = Number(localStorage.getItem("lastId")) || 0; // Persisted ID counter
    const setReminders = new Set();
    //let poupStatusActive = false;
    const popupQueue = [];
    let isPopupShowing = false;

    function playAlertSound() {
      const sound = document.getElementById('alert-sound');
      if (sound) {
        sound.play();
      }
    }

    // Ask for permission once
    if ('Notification' in window && Notification.permission !== 'granted') {
      Notification.requestPermission().then(permission => {
        if (permission === 'granted') {
          console.log("✅ Notification permission granted.");
        } else {
          console.log("❌ Notification permission denied.");
        }
      });
    }

    // Helper to show notification
    function showNotification(message) {
      if (Notification.permission === 'granted') {
        new Notification("Todo Reminder", { body: message });
      }
    }

    window.onload = function () {
      const stored = localStorage.getItem('todos');
      if (stored) {
        todos = JSON.parse(stored);
        renderTodos();
      }
      //console.log("Notification permission:", Notification.permission);

    };

    function saveTodos() {
      localStorage.setItem('todos', JSON.stringify(todos));
    }

    function addTodo() {
      const todo_Text = document.getElementById('todo-input').value.trim();
      const due_Date = document.getElementById('due-date').value;
      const _priority = document.getElementById('priority').value;

      if (text === '') return;

      ++id;
      localStorage.setItem("lastId", id);
      todos.push({
        userId: id,
        todoText: todo_Text,
        dueDate: due_Date,
        priority: _priority,
        completed: false,
        snoozedUntil: null,
        stopReminding: false
      });

      saveTodos();
      renderTodos();

      document.getElementById('todo-input').value = '';
      document.getElementById('due-date').value = '';
    }

    function renderTodos() {
      const list = document.getElementById('todo-list');
      list.innerHTML = '';

      const filter = document.getElementById('filter').value;
      const sort = document.getElementById('sort').value;

      let filtered = todos.filter(todo => {
        if (filter === 'completed') return todo.completed;
        if (filter === 'incomplete') return !todo.completed;
        if (filter === 'high') return todo.priority === "high";
        if (filter === 'medium') return todo.priority === "medium";
        if (filter === 'low') return todo.priority === "low";
        return true;
      });

      const order = { high: 1, medium: 2, low: 3 };
      if (sort === 'date') {
        filtered.sort((a, b) => new Date(a.dueDate || '9999-12-31') - new Date(b.dueDate || '9999-12-31'));
      } else if (sort === 'date-desc') {
        filtered.sort((a, b) => new Date(b.dueDate || '9999-12-31') - new Date(a.dueDate || '9999-12-31'));
      } else if (sort === 'priority') {
        filtered.sort((a, b) => order[a.priority] - order[b.priority]); // High - low
      } else if (sort === 'priority-desc') {
        filtered.sort((a, b) => order[b.priority] - order[a.priority]); // Low - high
      }

      filtered.forEach((todo) => {
        const li = document.createElement('li');
        li.className = 'fade-in';
        if (todo.completed) li.classList.add('completed');
        //let i = todos.indexOf(todo); - works but okay for small apps but not for production level as can create problem while serialization
        const now = new Date().getTime();
        const snoozedTime = todo.snoozedUntil ? new Date(todo.snoozedUntil).getTime() : 0;
        const dueDate = new Date(todo.dueDate);
        let isOverdue = !todo.completed && todo.dueDate && dueDate < now;

        li.innerHTML = `
        <input type="checkbox" data-action="toggle" data-id="${todo.id}" ${todo.completed ? 'checked' : ''}>
        <span class="todo-text">${todo.text}</span>
        <span class="meta priority-${todo.priority}">[${todo.priority}]</span>
        ${todo.dueDate ? `<span class="meta ${isOverdue ? 'overdue' : ''}">${isOverdue ? '🔴' : ''}Due: ${todo.dueDate}</span>` : ''}
        <button id="btnEdit" data-action="edit" data-id="${todo.id}">Edit</button>
        <button id="btnDelete" data-action="delete" data-id="${todo.id}">Delete</button>
      `;
        list.appendChild(li);

        // console.log(setReminders.has(todo.id) + " 1 " + todo.text);
        // console.log(todo.snoozedUntil + " " + todo.stopReminding);
        // console.log(todo.dueDate && !todo.completed && (!todo.snoozedUntil || new Date(todo.snoozedUntil).getTime() < Date.now()) && !(setReminders.has(todo.id)));
        //console.log((!todo.snoozedUntil || new Date(todo.snoozedUntil).getTime() < Date.now()));
        //console.log("last check-------------");
        // console.log("snoozedUntil:", snoozedTime, "| now:", Date.now(), "| comparison snoozedTime < Date.now():", snoozedTime < Date.now());
        // console.log("1st Snooze time snoozedTime > Date.now()", snoozedTime, " ", Date.now(), " - ", snoozedTime > Date.now());
        // Reminders (only if not completed)
        if (todo.dueDate && !todo.completed && !todo.stopReminding && snoozedTime < Date.now() && !(setReminders.has(todo.id))) {
          setReminders.add(todo.id);
          // console.log(setReminders.has(todo.id) + " 2 " + todo.text);
          const msUntilDue = dueDate - now;
          const earlyOffset = 10 * 60 * 1000; // 10 minutes before
          const afterInterval = 60 * 1000; // every 5 minutes after due

          // Reminder before due
          const msUntilEarly = msUntilDue - earlyOffset;
          if (msUntilEarly > 0) {
            setTimeout(() => {
              //alert(`🟡 "${todo.text}" 1 is due in 10 minutes!`);
              customAlertPopup(todo);
              showNotification(`"${todo.text}" is due in 10 minutes!`);
            }, msUntilEarly);
          }

          // Reminder at due time + after overdue loop
          if (msUntilDue > 0) {
            setTimeout(() => {
              customAlertPopup(todo);
              showNotification(`"${todo.text}" is now due!`);
     
                const intervalId = setInterval(() => {
                  const task = todos.find(t => t.id === todo.id);
                  
              const snoozedTime = task?.snoozedUntil ? new Date(task.snoozedUntil).getTime() : 0;
              const delay = snoozedTime > Date.now()  ? snoozedTime - Date.now() : 0;

                  if (!task || task.completed || task.stopReminding || snoozedTime > Date.now()) {
                    clearInterval(intervalId);
                    setReminders.delete(task.id);
                  }

                  playAlertSound();
                  customAlertPopup(task);
                  showNotification(`"${task.text}" is still overdue!`);
                }, afterInterval);
            
            }, msUntilDue);
          } else {
            //console.log("snoozedTime ", snoozedTime, " delay ", delay, " Date.now() ", Date.now(), " Greater check ", snoozedTime > Date.now());
            
              const intervalId = setInterval(() => {
                const task = todos.find(t => t.id === todo.id);
                const snoozedTime = task?.snoozedUntil ? new Date(task.snoozedUntil).getTime() : 0;
              const delay = snoozedTime > Date.now() ? snoozedTime - Date.now() : 0;
                if (!task || task.completed || task.stopReminding || snoozedTime > Date.now()) {
                  clearInterval(intervalId);
                  setReminders.delete(task.id);
                }

                playAlertSound();
                customAlertPopup(task);
                showNotification(`"${task.text}" is still overdue!`);
              }, afterInterval);
            
          }
        }
      });
    }

    document.getElementById('todo-list').addEventListener('click', function (ev) {
      const id = ev.target.dataset.id;
      const action = ev.target.dataset.action;

      if (!id) return;

      const index = todos.findIndex(todo => todo.id == id);
      if (index === -1) return;

      if (ev.target.type === 'checkbox' && action === 'toggle') {
        todos[index].completed = ev.target.checked;
        if (todos[index].completed) {
          setReminders.delete(todos[index].id); // setReminders.delete(todo.id);
        }

        saveTodos();
        renderTodos();

        /* If don't want render to run after checkbox toggling instead the event handler itself handle the required processing
        const li = ev.target.closest('li');
        if (todos[index].completed) {
          li.classList.add('completed');
        } else {
          li.classList.remove('completed');
        }
        const now = new Date();
        const dueDate = new Date(todos[index].dueDate);
        let isOverdue = !todos[index].completed && todos[index].dueDate && dueDate < now;
        let dueDateSpan = li.querySelector(".due-date");
        if (isOverdue) {
          dueDateSpan.classList.add('overdue');
        }
        else {
          dueDateSpan.classList.remove('overdue');
        } */

        return; // Exit early tom avoid edit/delete triggering
      }

      if (action === 'delete') {
        todos.splice(index, 1);
      } else if (action === 'edit') {
        const newText = prompt('Edit task:', todos[index].text);
        if (newText !== null && newText.trim() !== '') {
          todos[index].text = newText.trim();
        }
      }

      saveTodos();
      renderTodos();
    });

    function customAlertPopup(todo) {
      popupQueue.push(todo);
      processNextPopup();
    }

    function processNextPopup() {
      if (isPopupShowing || popupQueue.length === 0) return;

      const todo = popupQueue.shift();
      const popup = document.getElementById('reminder-popup');
      document.getElementById('popup-text').textContent = `🔴 "${todo.text}" is overdue!`;
      popup.style.display = 'flex';
      isPopupShowing = true;

      document.getElementById('popup-snooze').onclick = () => {
        todo.snoozedUntil = new Date(Date.now() + 2 * 60 * 1000);
        // console.log("Snoozed time set ***********************", new Date(Date.now() + 2 * 60 * 1000));
        //setReminders.delete(todo.id);
        popup.style.display = 'none';
        saveTodos();
        renderTodos();
        isPopupShowing = false;
        processNextPopup();
      };

      document.getElementById('popup-stop').onclick = () => {
        todo.stopReminding = true;
        popup.style.display = 'none';
        saveTodos();
        renderTodos();
        isPopupShowing = false;
        processNextPopup();
      };

      document.getElementById('popup-close').onclick = () => {
        popup.style.display = 'none';
        isPopupShowing = false;
        processNextPopup();
      };
    }
